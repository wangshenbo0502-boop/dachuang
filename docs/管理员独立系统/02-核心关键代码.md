## 1. 严格命令 DTO

目标：`backend/app/internal_admin/schemas.py`。管理后端须有契约一致的输入模型，通过契约测试保证一致，不直接 import 本文件。

```python
from typing import Annotated, Literal
from pydantic import BaseModel, ConfigDict, Field, StrictBool, field_validator

Version = Annotated[int, Field(strict=True, ge=0)]

class CommandInput(BaseModel):
    model_config = ConfigDict(extra="forbid")
    expected_version: Version
    reason: str = Field(min_length=5, max_length=500)

    @field_validator("reason", mode="before")
    @classmethod
    def trim_reason(cls, value: object) -> object:
        return value.strip() if isinstance(value, str) else value

class AccountStatusCommand(CommandInput):
    is_active: StrictBool

class JobModerationCommand(CommandInput):
    moderation_status: Literal["allowed", "blocked"]
```

禁止把 `is_active="false"`、`expected_version=true`、`actor_id` 或 `role` 静默转换为合法写入。FastAPI 的校验异常处理器应删除错误对象中的原始 input，尤其是登录/改密请求。

## 2. 独立管理模型与 RBAC

目标：`admin-backend/admin_app/models.py` 的身份部分。`admin_app.db.Base` 必须是新管理库的 DeclarativeBase，不是业务库 Base。其他管理表按契约实现。

```python
from uuid import uuid4
from sqlalchemy import Boolean, ForeignKey, Integer, String

from admin_app.db import Base

class AdminAccount(Base):
```

目标：`admin-backend/admin_app/permissions.py`。

```python
    "overview:read", "accounts:read", "recruiters:read", "jobs:read",
    "applications:read", "commands:read", "audit:read",
```

目标：`admin-backend/admin_app/security.py` 的会话鉴权与签发函数。它依赖两个需在 P1 创建的模块：

- `admin_app.db.get_db()`：只连接 `ADMIN_DATABASE_URL` 的每请求 Session 生成器，在 finally 中 close。
- `admin_app.config.get_settings()`：返回 cookie_name、cookie_secure、admin_origin；production 必须 Secure，明确读取管理目录配置，禁止搜索并加载父目录业务 `.env`。

```python
from dataclasses import dataclass
import hashlib
import hmac
import secrets
import time
from fastapi import Depends, HTTPException, Request, Response
from sqlalchemy.orm import Session
from admin_app.config import get_settings
from admin_app.db import get_db
from admin_app.models import AdminAccount, AdminSession
from admin_app.permissions import ROLE_PERMISSIONS

@dataclass(frozen=True)
class AdminPrincipal:
    id: str
    role: str
    permissions: frozenset[str]
    session_hash: str
    csrf_token: str

def token_hash(value: str) -> str:
    return hashlib.sha256(value.encode("utf-8")).hexdigest()

def issue_session(db: Session, account: AdminAccount) -> tuple[str, str]:
    # 在调用方登录事务内执行，调用方提交后才能设置 Cookie。
    raw_token = secrets.token_urlsafe(32)
    csrf = secrets.token_urlsafe(32)
    db.add(AdminSession(
        token_hash=token_hash(raw_token), admin_id=account.id,
        version=account.session_version, csrf_token=csrf,
        expires_at=int(time.time()) + 8 * 60 * 60,
    ))
    db.flush()
    return raw_token, csrf

def set_session_cookie(response: Response, raw_token: str) -> None:
    cfg = get_settings()
    response.set_cookie(
        key=cfg.cookie_name, value=raw_token,
        httponly=True, secure=cfg.cookie_secure, samesite="strict",
        path="/admin-api/v1", max_age=8 * 60 * 60,
    )
    response.headers["Cache-Control"] = "no-store"

def require_origin(request: Request) -> None:
    if request.headers.get("origin") != get_settings().admin_origin:
        raise HTTPException(403, "请求来源不允许")

def require_admin(
    request: Request, db: Session = Depends(get_db),
) -> AdminPrincipal:
    raw = request.cookies.get(get_settings().cookie_name, "")
    if not raw or len(raw) > 256:
        raise HTTPException(401, "请先登录管理员系统")
    row = db.get(AdminSession, token_hash(raw))
    if row is None or row.revoked_at is not None or row.expires_at <= time.time():
        raise HTTPException(401, "管理员会话已失效")
    account = db.get(AdminAccount, row.admin_id)
    if (account is None or not account.is_active
            or account.session_version != row.version):
        raise HTTPException(401, "管理员会话已失效")
    permissions = ROLE_PERMISSIONS.get(account.role)
    if permissions is None:
        raise HTTPException(403, "管理员角色无效")
    if request.method not in {"GET", "HEAD", "OPTIONS"}:
        require_origin(request)
        supplied = request.headers.get("x-csrf-token", "")
        if not supplied or not hmac.compare_digest(supplied, row.csrf_token):
            raise HTTPException(403, "请求校验失败")
    return AdminPrincipal(
        id=account.id, role=account.role, permissions=permissions,
        session_hash=row.token_hash, csrf_token=row.csrf_token,
    )

def require_permission(permission: str):
    def dependency(
        principal: AdminPrincipal = Depends(require_admin),
    ) -> AdminPrincipal:
        if permission not in principal.permissions:
            raise HTTPException(403, "没有该操作权限")
        return principal
    return dependency
```

HTTPException 在管理应用集中转为约定的 envelope，保持原 HTTP 状态。不要将 403 捕获后改为 200。`/auth/me` 的响应模型只暴露 id、username、role、permissions、csrf_token，不直接序列化 AdminPrincipal。

登录过程必须完成 Argon2 验证、当前账号状态和限流检查，再在事务中创建会话、更新登录审计。不存在的账号也进行固定 dummy hash 校验以降低明显的响应时间差异。登录创建会话与并发禁用/改密须按同一账号版本协调；新会话版本不匹配时下次请求拒绝。

## 3. 服务请求的签名和校验

管理端使用 HTTPX，服务 JWT 仅在后端生成。冻结正文采用以下规范；**首次持久化后重试必须读取已保存的 bytes，不重新生成**。

目标：`admin-backend/admin_app/clients/business.py` 的协议函数。

```python
import hashlib
import json
import time
from uuid import uuid4
import jwt

def encode_body(body: dict) -> bytes:
    return json.dumps(
        body, ensure_ascii=False, sort_keys=True, separators=(",", ":")
    ).encode("utf-8")

def request_digest(method: str, target: str, body: bytes, command_id: str) -> str:
    prefix = f"{method.upper()}\n{target}\n{command_id}\n".encode("utf-8")
    return hashlib.sha256(prefix + body).hexdigest()

def sign_call(
    *, secret: str, actor_id: str, scope: str, method: str,
    target: str, body: bytes = b"", command_id: str = "",
) -> str:
    if len(secret.encode("utf-8")) < 32:
        raise ValueError("内部调用密钥未正确配置")
    now = int(time.time())
    return jwt.encode({
        "iss": "dachuang-admin-service",
        "aud": "dachuang-business-internal",
        "sub": "admin-service", "actor": actor_id, "scope": scope,
        "req": request_digest(method, target, body, command_id),
        "iat": now, "exp": now + 30, "jti": str(uuid4()),
    }, secret, algorithm="HS256")
```

目标：`backend/app/internal_admin/security.py`。其中 request_digest 在业务侧本地实现同样的纯协议算法，用固定测试向量验证一致；不能从管理工程 import。密钥由原 config 新增字段读取，接口关闭时无需配置。

```python
from dataclasses import dataclass
import hashlib
import hmac
from uuid import UUID
import jwt
from fastapi import Depends, HTTPException, Request
from fastapi.security import HTTPAuthorizationCredentials, HTTPBearer
from app.config import get_settings

bearer = HTTPBearer(auto_error=False)

@dataclass(frozen=True)
class ServiceActor:
    id: str
    request_hash: str
    scope: str

def request_digest(method: str, target: str, body: bytes, command_id: str) -> str:
    prefix = f"{method.upper()}\n{target}\n{command_id}\n".encode("utf-8")
    return hashlib.sha256(prefix + body).hexdigest()

def require_service_scope(scope: str):
    async def dependency(
        request: Request,
        credentials: HTTPAuthorizationCredentials | None = Depends(bearer),
    ) -> ServiceActor:
        if credentials is None or credentials.scheme.lower() != "bearer":
            raise HTTPException(401, "需要服务凭证")
        cfg = get_settings()
        try:
            claims = jwt.decode(
                credentials.credentials, cfg.INTERNAL_ADMIN_SERVICE_SECRET,
                algorithms=["HS256"], audience="dachuang-business-internal",
                issuer="dachuang-admin-service",
                options={"require": [
                    "iss", "aud", "sub", "actor", "scope", "req", "iat", "exp", "jti"
                ]},
            )
            actor_id = str(UUID(claims["actor"]))
            UUID(claims["jti"])
            valid_lifetime = (
                type(claims["iat"]) is int and type(claims["exp"]) is int
                and 0 < claims["exp"] - claims["iat"] <= 30
            )
            if claims["sub"] != "admin-service" or not valid_lifetime:
                raise ValueError("invalid claims")
            if not isinstance(claims["req"], str):
                raise ValueError("invalid digest")
        except (jwt.InvalidTokenError, ValueError, TypeError, AttributeError):
            raise HTTPException(401, "服务凭证无效") from None
        if claims["scope"] != scope:
            raise HTTPException(403, "服务权限不足")
        raw_path = request.scope.get("raw_path", request.url.path.encode("ascii"))
        query = request.scope.get("query_string", b"")
        target = raw_path.decode("ascii") + ("?" + query.decode("ascii") if query else "")
        command_id = request.headers.get("idempotency-key", "")
        expected = request_digest(request.method, target, await request.body(), command_id)
        if not hmac.compare_digest(expected, claims["req"]):
            raise HTTPException(403, "服务请求内容不匹配")
        return ServiceActor(actor_id, expected, scope)
    return dependency
```

服务 JWT 验签异常不能返回 token、secret 或原始库异常。生产启用前检查内部密钥与原业务 JWT_SECRET 不相同；配置层校验开关、长度及路径来源。request_id 限制长度和字符集，用于追踪，不承担鉴权。

HTTPX 客户端在管理应用 lifespan 创建并关闭，`trust_env=False`、`follow_redirects=False`，固定服务 base_url，超时 `httpx.Timeout(10.0, connect=2.0, pool=2.0)`。不得允许浏览器传 target URL，避免变成任意代理。签名基于 HTTPX 已构造 Request 的 raw_path，确保 query 编码与实际发送一致：

```python
import httpx

async def send_frozen_command(
    client: httpx.AsyncClient, *, path: str, raw_body: bytes,
    command_id: str, actor_id: str, scope: str, secret: str,
) -> httpx.Response:
    # path 只能由服务端固定映射和已校验的整数资源 ID 构造。
    request = client.build_request(
        "POST", path, content=raw_body,
        headers={"Content-Type": "application/json", "Idempotency-Key": command_id},
    )
    target = request.url.raw_path.decode("ascii")
    request.headers["Authorization"] = "Bearer " + sign_call(
        secret=secret, actor_id=actor_id, scope=scope, method="POST",
        target=target, body=raw_body, command_id=command_id,
    )
    return await client.send(request)
```

该发送函数由命令服务调用。它不自动 retry、不把 timeout 转换为失败、不处理本地命令持久化；这些由第 5 节明确的命令服务负责。GET 客户端同样先 build_request 再签名，可最多重试一次网络失败；写请求只走持久化命令机制。

## 4. 业务凭据、幂等与条件更新

目标：`backend/app/models/admin_receipt.py`。业务凭据既是去重记录，也是管理变更在业务库的权威审计记录；只能通过专用查询接口读取。


前提：给现有 Account 添加 `management_version` 并完成补列迁移，模型注册必须包含 BusinessAdminReceipt。以下代码使用**独立 Session**，不复用完成认证查询的请求 Session，也不调用会自行 commit 的原业务服务方法。

目标：`backend/app/internal_admin/commands.py` 的账号状态命令参考。

```python
from uuid import UUID
from fastapi import HTTPException
from sqlalchemy import select, update
from sqlalchemy.exc import IntegrityError
from app.database.session import get_session_factory
from app.models.account import Account
from app.models.admin_receipt import BusinessAdminReceipt
from app.internal_admin.schemas import AccountStatusCommand
from app.internal_admin.security import ServiceActor

def envelope(code: int, message: str, data: dict | None, command_id: str) -> dict:
    return {"code": code, "message": message, "data": data, "request_id": command_id}

def replay(receipt: BusinessAdminReceipt, actor: ServiceActor) -> tuple[int, dict]:
    if receipt.actor_id != actor.id or receipt.request_hash != actor.request_hash:
        raise HTTPException(409, "幂等键已用于其他请求")
    return receipt.http_status, dict(receipt.response_json)

def change_account_status(
    *, account_id: int, command_id: str, body: AccountStatusCommand,
    actor: ServiceActor,
) -> tuple[int, dict]:
    # HTTP 层也应使用 UUID 类型校验 header；这里再次约束函数输入。
    command_id = str(UUID(command_id))
    factory = get_session_factory()
    try:
        with factory.begin() as db:
            old = db.get(BusinessAdminReceipt, command_id)
            if old is not None:
                return replay(old, actor)
            receipt = BusinessAdminReceipt(
                command_id=command_id, actor_id=actor.id,
                operation="accounts:status_write", resource_id=str(account_id),
                request_hash=actor.request_hash, reason=body.reason,
                before_json={}, after_json={}, http_status=500, response_json={},
            )
            db.add(receipt)
            # 先竞争唯一键；并发重复请求必须先等待这笔事务的结果。
            db.flush()
            current = db.execute(select(
                Account.is_active, Account.management_version
            ).where(Account.id == account_id)).mappings().one_or_none()
            if current is None:
                status, payload = 404, envelope(6404, "账号不存在", None, command_id)
            elif current["management_version"] != body.expected_version:
                status, payload = 409, envelope(6409, "账号已变化，请刷新", None, command_id)
            elif current["is_active"] == body.is_active:
                status, payload = 409, envelope(6409, "账号已处于目标状态", None, command_id)
            else:
                changed = db.execute(update(Account).where(
                    Account.id == account_id,
                    Account.management_version == body.expected_version,
                    Account.is_active == current["is_active"],
                ).values(
                    is_active=body.is_active,
                    token_version=Account.token_version + 1,
                    management_version=Account.management_version + 1,
                ), execution_options={"synchronize_session": False})
                if changed.rowcount != 1:
                    status, payload = 409, envelope(6409, "账号已变化，请刷新", None, command_id)
                else:
                    receipt.before_json = {
                        "is_active": current["is_active"],
                        "management_version": current["management_version"],
                    }
                    receipt.after_json = {
                        "is_active": body.is_active,
                        "management_version": body.expected_version + 1,
                    }
                    status, payload = 200, envelope(0, "账号状态已更新", {
                        "command_id": command_id, "id": account_id,
                        **receipt.after_json,
                    }, command_id)
            receipt.http_status = status
            receipt.response_json = payload
            # 离开 begin 上下文才提交，提交失败将抛出，不能返回成功。
        return status, payload
    except IntegrityError:
        # 可能是 command_id 竞争，也可能是其他约束错误；只恢复确实存在的凭据。
        with factory() as db:
            committed = db.get(BusinessAdminReceipt, command_id)
            if committed is None:
                raise
            return replay(committed, actor)
```

SQLite 的锁超时、数据库断连等异常不应强行吞成 409；整个事务回滚，由上层返回可重试的服务错误，管理端按 unknown 处理并查询原 ID。生产目标数据库必须验证唯一键竞争、条件更新及回滚行为。

目标：`backend/app/internal_admin/router.py` 的路由装配示例。

### 岗位治理的必要条件

岗位限制命令采用上述同一种凭据事务机制，在本地业务模块中实现，不跨应用复用 ORM。关键 SQL 条件如下；此函数不提交，由持有凭据的外层事务负责提交。

```python
from sqlalchemy import update
from sqlalchemy.orm import Session
from app.models.recruitment import RecruitmentJob

def update_moderation(
    db: Session, *, job_id: int, expected_version: int, target: str,
) -> int:
    if target not in {"allowed", "blocked"}:
        raise ValueError("invalid moderation status")
    result = db.execute(update(RecruitmentJob).where(
        RecruitmentJob.id == job_id,
        RecruitmentJob.management_version == expected_version,
        RecruitmentJob.moderation_status != target,
    ).values(
        moderation_status=target,
        management_version=RecruitmentJob.management_version + 1,
    ), execution_options={"synchronize_session": False})
    return result.rowcount
```

必须同时改动 `job_query()`、`public_job()` 和 `set_job_status()` 等真实路径：公开列表/详情/投递要求 `moderation_status='allowed'`；发布更新本身也包含该条件及当前版本。编辑、发布和下架统一原子增加 management_version。只在 Python 里先检查后无条件更新，会留下并发绕过窗口。

原 `serialize()` 自动输出新列，实施时需审查原岗位响应，改用明确允许字段；治理理由只存凭据。招聘者必要状态 DTO 可含 moderation_status 和 management_version，求职者不需要治理操作细节。

## 5. 管理命令服务：必须实现的状态处理

目标：`admin-backend/admin_app/services/commands.py`。下表是实现契约，不能用一个 HTTP 转发函数替代。

| 函数 | 输入 | 必须执行/输出 |
|---|---|---|
| `accept_command(principal, operation, resource_id, body, key)` | 已授权 principal、严格 DTO、UUID key | 管理库事务插入冻结 body、请求摘要、pending 意图和受理审计；重复键先比较 actor/方法/路径/body，完全相同才返回已有命令 |
| `dispatch_command(command_id)` | 已持久化命令 | 加命令调度互斥；取冻结请求；不持有管理库事务等待网络；签名并发送 |
| `record_business_result(command_id, status, payload)` | 已验证业务响应 | 核验 command_id、envelope 和预期 DTO；200/code=0 为 succeeded；已知业务 4xx 凭据为 rejected；同事务更新命令与终态审计 |
| `mark_unknown(command_id, reason_code)` | 网络/5xx/非法响应 | 状态 pending→unknown；不得覆盖已存在 succeeded/rejected；reason_code 为枚举，不保存含密钥的异常字符串 |
| `reconcile_command(command_id, principal)` | 有查看权限的命令 | 查询业务凭据；查到终态则幂等落盘；404/网络失败保留待确认 |
| `retry_command(command_id, principal)` | 本人且仍有原写权限，或 super_admin | 先查凭据；仍待确认时原 key、原 body 重发；actor 使用记录中的原操作者，审计另外记录此次重试操作者 |

**恢复扫描只查询结果，不自动重发未确认写入。** 这样管理员被禁用或角色降级后不会产生新的后台写请求。需要重发时由仍具权限的操作者显式重试；服务 JWT 已发出且仍在 30 秒有效期内的在途命令可能完成。

每个终态审计以 `(command_id, event_kind)` 建唯一约束，避免恢复扫描重复产生日志。管理端对自己账号、权限、会话的写操作全部在管理库本地事务完成，无需业务 HTTP。

目标：`admin-frontend/src/api/http.ts`。

```typescript
import axios from 'axios';

export interface ApiEnvelope<T> {
  code: number;
  message: string;
  data: T;
  request_id: string;
}

export class AdminApiError extends Error {
  constructor(message: string, public status: number, public code?: number) {
    super(message);
    this.name = 'AdminApiError';
  }
}

let csrfToken = '';
export function setCsrfToken(value: string): void { csrfToken = value; }

export const http = axios.create({
  baseURL: '/admin-api/v1', timeout: 15000, withCredentials: true,
});

http.interceptors.request.use((config) => {
  if (!['get', 'head', 'options'].includes((config.method ?? 'get').toLowerCase())) {
    if (csrfToken) config.headers.set('X-CSRF-Token', csrfToken);
  }
  return config;
});

export async function apiData<T>(
  request: Promise<{ data: ApiEnvelope<T>; status: number }>,
): Promise<T> {
  try {
    const response = await request;
    if (response.data.code !== 0) {
      throw new AdminApiError(response.data.message, response.status, response.data.code);
    }
    return response.data.data;
  } catch (error) {
    if (error instanceof AdminApiError) throw error;
    if (axios.isAxiosError<ApiEnvelope<unknown>>(error)) {
      throw new AdminApiError(
        error.response?.data?.message ?? '管理服务暂不可用',
        error.response?.status ?? 0, error.response?.data?.code,
      );
    }
    throw error;
  }
}
```

错误拦截器不一律清除登录：401 代表会话问题，403 代表权限，503/网络失败代表可用性。页面数据请求统一经应用错误处理入口，遇当前会话的 401 再清理 Store 并导航，禁止让旧请求覆盖新登录状态。

目标：`admin-frontend/src/stores/auth.ts` 的核心状态。Me 类型应最终由 OpenAPI 生成；本例把形状列出便于阅读。

```typescript
import { computed, ref } from 'vue';
import { defineStore } from 'pinia';
import { AdminApiError, apiData, http, setCsrfToken } from '@/api/http';

interface Me {
  admin: { id: string; username: string; role: string };
  permissions: string[];
  csrf_token: string;
}

export const useAdminAuth = defineStore('admin-auth', () => {
  const me = ref<Me | null>(null);
  let flight: Promise<Me | null> | null = null;
  let epoch = 0;
  const permissions = computed(() => new Set(me.value?.permissions ?? []));

  function clear(): void {
    epoch += 1;
    me.value = null;
    setCsrfToken('');
    flight = null;
  }

  function ensureSession(): Promise<Me | null> {
    if (flight) return flight;
    const started = epoch;
    const task = apiData<Me>(http.get('/auth/me')).then((value) => {
      if (started !== epoch) return me.value;
      me.value = value;
      setCsrfToken(value.csrf_token);
      return value;
    }).catch((error: unknown) => {
      if (started !== epoch) return me.value;
      if (error instanceof AdminApiError && error.status === 401) {
        clear();
        return null;
      }
      throw error;
    }).finally(() => {
      if (flight === task) flight = null;
    });
    flight = task;
    return task;
  }

  async function login(username: string, password: string): Promise<void> {
    clear();
    const started = epoch;
    const value = await apiData<Me>(http.post('/auth/login', { username, password }));
    if (started !== epoch) return;
    me.value = value;
    setCsrfToken(value.csrf_token);
  }

  async function logout(): Promise<void> {
    // 只有服务端确认吊销或确认会话已失效，才宣称退出成功。
    try {
      await apiData<null>(http.post('/auth/logout'));
    } catch (error) {
      if (!(error instanceof AdminApiError && error.status === 401)) throw error;
    }
    clear();
  }

  return { me, permissions, clear, ensureSession, login, logout };
});
```

每次受保护路由转换刷新 `/auth/me`，并发导航复用一个请求；后端仍在每次 API 请求实时检查权限。登录/退出按钮需要单次提交锁，避免并发 Set-Cookie 响应竞争。退出遇网络错误应显示“未能确认退出，请重试”，不能只清内存就宣称 Cookie 已失效。

```typescript
export function safeAdminRedirect(raw: unknown): string {
  if (typeof raw !== 'string' || !raw.startsWith('/') || raw.startsWith('//')) {
    return '/overview';
  }
  if (raw.includes('\\') || /[\u0000-\u001f\u007f]/.test(raw)) return '/overview';
  try {
    const url = new URL(raw, window.location.origin);
    const resolved = router.resolve(url.pathname + url.search + url.hash);
    if (url.origin !== window.location.origin || !resolved.matched.length) return '/overview';
    if (['/login', '/403', '/unavailable'].includes(url.pathname)) return '/overview';
    if (resolved.name === 'not-found') return '/overview';
    return url.pathname + url.search + url.hash;
  } catch {
    return '/overview';
  }
}
```

## 7. 前端账号操作：幂等键与待确认状态

目标：`admin-frontend/src/api/accounts.ts`。该例展示协议，实际接口类型应从契约生成。


页面必须在第一次发送前生成 `crypto.randomUUID()` 并保存冻结的请求对象。浏览器到管理后端的请求也可能超时；即使没收到 command_id，仍能用这个 key 查询，或重复提交原正文。查询返回 404 时不得立刻创建新键。

以下为 `CommandConfirmDialog.vue` 的提交逻辑示例，依赖 props.account、reason、emits 等由实际组件声明，重点是冻结请求与重试语义：

```typescript
import { ref } from 'vue';
import { AdminApiError } from '@/api/http';
import { changeAccountStatus, readCommand } from '@/api/accounts';
import type { AccountStatusInput, CommandResult } from '@/api/accounts';

interface FrozenAction {
  key: string;
  accountId: number;
  body: Readonly<AccountStatusInput>;
}

const pending = ref<FrozenAction | null>(null);
const busy = ref(false);
const message = ref('');

async function send(action: FrozenAction): Promise<CommandResult | null> {
  if (busy.value) return null;
  busy.value = true;
  try {
    const result = await changeAccountStatus(action.accountId, action.body, action.key);
    message.value = result.state === 'succeeded' ? '操作已完成'
      : result.state === 'rejected' ? (result.result?.response.message ?? '操作被拒绝，请查看原因')
      : `操作结果待确认，编号：${action.key}`;
    if (['succeeded', 'rejected'].includes(result.state)) pending.value = null;
    return result;
  } catch (error) {
    // 明确的本地鉴权/校验错误与传输结果不明分开处理。
    if (error instanceof AdminApiError && [401, 403, 409, 422].includes(error.status)) {
      message.value = error.message;
      // 保留 key 便于核查；409 不能静默换键重发。
    } else {
      message.value = `操作结果待确认，编号：${action.key}。请查询原操作。`;
    }
    return null;
  } finally {
    busy.value = false;
  }
}

function prepare(accountId: number, version: number, active: boolean, reason: string) {
  if (pending.value) throw new Error('请先确认上一笔操作结果');
  const trimmed = reason.trim();
  if (trimmed.length < 5 || trimmed.length > 500) throw new Error('理由需为 5～500 字');
  pending.value = {
    key: crypto.randomUUID(), accountId,
    body: Object.freeze({ is_active: active, expected_version: version, reason: trimmed }),
  };
  return pending.value;
}

async function queryPending(): Promise<CommandResult | null> {
  if (!pending.value) return null;
  const result = await readCommand(pending.value.key);
  if (['succeeded', 'rejected'].includes(result.state)) pending.value = null;
  return result;
}
```
